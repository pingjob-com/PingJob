STEP 2: DATABASE SETUP - RUN THIS SQL
--------------------------------------
Execute this SQL script on your database to ensure required tables exist:

-- Connections table for user connections
CREATE TABLE IF NOT EXISTS "connections" (
"id" serial PRIMARY KEY NOT NULL,
"requester_id" varchar NOT NULL,
"requested_id" varchar NOT NULL,
"status" varchar DEFAULT 'pending',
"created_at" timestamp DEFAULT now(),
CONSTRAINT "connections_requester_id_users_id_fk"
FOREIGN KEY ("requester_id") REFERENCES "users"("id") ON DELETE CASCADE,
CONSTRAINT "connections_requested_id_users_id_fk"
FOREIGN KEY ("requested_id") REFERENCES "users"("id") ON DELETE CASCADE
);

-- Messages table for user messaging
CREATE TABLE IF NOT EXISTS "messages" (
"id" serial PRIMARY KEY NOT NULL,
"sender_id" varchar NOT NULL,
"receiver_id" varchar NOT NULL,
"content" text NOT NULL,
"sent_at" timestamp DEFAULT now(),
"read_at" timestamp,
"is_read" boolean DEFAULT false,
CONSTRAINT "messages_sender_id_users_id_fk"
FOREIGN KEY ("sender_id") REFERENCES "users"("id") ON DELETE CASCADE,
CONSTRAINT "messages_receiver_id_users_id_fk"
FOREIGN KEY ("receiver_id") REFERENCES "users"("id") ON DELETE CASCADE
);

-- Performance indexes for faster queries
CREATE INDEX IF NOT EXISTS "idx_connections_requester" ON "connections"("requester_id");
CREATE INDEX IF NOT EXISTS "idx_connections_requested" ON "connections"("requested_id");
CREATE INDEX IF NOT EXISTS "idx_connections_status" ON "connections"("status");
CREATE INDEX IF NOT EXISTS "idx_messages_sender" ON "messages"("sender_id");
CREATE INDEX IF NOT EXISTS "idx_messages_receiver" ON "messages"("receiver_id");
CREATE INDEX IF NOT EXISTS "idx_messages_read" ON "messages"("is_read");
